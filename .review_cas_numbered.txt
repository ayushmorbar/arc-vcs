  14: const SMALL_OBJECT_THRESHOLD: u64 = 4096;
  15: static TEMP_FILE_COUNTER: AtomicU64 = AtomicU64::new(0);
  16: 
  17: /// Errors returned by the standalone CAS engine.
  18: #[derive(Debug, Error)]
  19: pub enum CasError {
  20:     /// Filesystem I/O failure.
  21:     #[error("I/O error: {0}")]
  22:     Io(#[from] std::io::Error),
  23: }
  24: 
  25: /// Files smaller than one OS page (4 KiB) are read into a heap buffer so the
  26: /// kernel does not need to manage a very short-lived page-table entry.
  27: /// Larger blobs are returned as a memory-mapped slice, avoiding a copy
  28: /// into the process heap entirely.
  29: ///
  30: /// Callers access the bytes via `Deref<Target = [u8]>`; the distinction is
  31: /// fully transparent to them.
  32: pub enum CasBytes {
  33:     /// Heap-allocated bytes for small blobs (< 4 096 bytes).
  34:     ///
  35:     /// `Bytes` preserves zero-copy handoff into downstream streaming paths.
  36:     Owned(Bytes),
  37:     /// Memory-mapped file for large blobs (≥ 4 096 bytes).
  38:     Mapped(Mmap),
  39: }
  40: 
  41: impl std::ops::Deref for CasBytes {
  42:     type Target = [u8];
  43:     #[inline]
  44:     fn deref(&self) -> &[u8] {
  45:         match self {
  46:             CasBytes::Owned(v) => v,
  47:             CasBytes::Mapped(m) => m,
  48:         }
  49:     }
  50: }
  51: 
  52: impl Default for CasBytes {
  53:     #[inline]
  54:     fn default() -> Self {
  55:         CasBytes::Owned(Bytes::new())
  56:     }
  57: }
  58: 
  59: impl AsRef<[u8]> for CasBytes {
  60:     #[inline]
  61:     fn as_ref(&self) -> &[u8] {
  62:         self
  63:     }
  64: }

  66: /// Content-addressable object store backed by BLAKE3 hashes and `bincode`
  67: /// serialization.
  68: ///
  69: /// Objects are persisted at `.arc/store/{hex[0..2]}/{hex[2..]}` ensuring
  70: /// automatic deduplication — identical content always maps to the same path.
  71: pub struct ObjectStore {
  72:     root: PathBuf,
  73: }
  74: 
  75: impl ObjectStore {
  76:     /// Create a new `ObjectStore` rooted at `root/.arc`.
  77:     pub fn new(root: impl AsRef<Path>) -> Self {
  78:         Self {
  79:             root: root.as_ref().join(".arc"),
  80:         }
  81:     }
  82: 
  83:     /// Derive the on-disk path for a given BLAKE3 hash.
  84:     ///
  85:     /// Layout: `<root>/.arc/store/{first_2_hex_chars}/{remaining_hex_chars}`
  86:     fn object_path(&self, hash: &Blake3Hash) -> PathBuf {
  87:         let hex = hex_encode(hash);
  88:         self.root.join("store").join(&hex[..2]).join(&hex[2..])
  89:     }
  90: 
  91:     /// Persist an object under its precomputed BLAKE3 key.
  92:     ///
  93:     /// Semantic guarantee: storage is content-addressed by caller-supplied
  94:     /// BLAKE3 keys, and duplicate writes under an existing path are skipped.
  95:     ///
  96:     /// This is a local best-effort dedup behavior, not a crash-atomic
  97:     /// multi-writer protocol.
  98:     ///
  99:     /// Durability note: this method performs a direct file write and does not
 100:     /// issue an explicit fsync. It guarantees deterministic path placement and
 101:     /// dedup semantics, but power-loss durability must be provided by a higher
 102:     /// durability layer if required.
 103:     #[instrument(skip_all)]
 104:     pub fn write_object(&self, hash: &Blake3Hash, bytes: &[u8]) -> Result<Blake3Hash, CasError> {
 105:         let path = self.object_path(hash);
 106:         write_once_atomic(&path, bytes)?;
 107:         Ok(*hash)
 108:     }
 109: 
 110:     /// Read an object back from CAS by BLAKE3 key.
 111:     ///
 112:     /// Files < 4 096 bytes are read into a `Vec<u8>`; larger objects are
 113:     /// memory-mapped to avoid a heap copy.
 114:     #[instrument(skip_all)]
 115:     pub fn read_object(&self, hash: &Blake3Hash) -> Result<CasBytes, CasError> {
 116:         let path = self.object_path(hash);
 117:         read_cas_bytes(&path)
 118:     }
 119: 
 120:     /// Derive the on-disk path for a raw blob in `.arc/blobs/{hex(hash)}`.
 121:     fn blob_path(&self, hash: &Blake3Hash) -> PathBuf {
 122:         self.root.join("blobs").join(hex_encode(hash))
 123:     }
 124: 
 125:     /// Persist raw bytes as a content-addressed blob.
 126:     ///
 127:     /// Returns the BLAKE3 hash of the content (the blob's storage key).
 128:     /// If the blob already exists the write is skipped.
 129:     #[instrument(skip_all)]
 130:     pub fn write_blob(&self, bytes: &[u8]) -> Result<Blake3Hash, CasError> {
 131:         let hash: Blake3Hash = *blake3::hash(bytes).as_bytes();
 132:         let path = self.blob_path(&hash);
 133:         write_once_atomic(&path, bytes)?;
 134:         Ok(hash)
 135:     }
 136: 
 137:     /// Persist raw bytes as a content-addressed blob and return a typed
 138:     /// identifier for cross-crate APIs.
 139:     ///
 140:     /// This wrapper keeps type boundaries explicit (`BlobId` instead of
 141:     /// bare hash arrays) while preserving the same CAS write path.
 142:     #[instrument(skip_all)]
 143:     pub fn write_blob_typed(&self, bytes: &[u8]) -> Result<BlobId, CasError> {
 144:         self.write_blob(bytes).map(BlobId::from)
 145:     }
 146: 
 147:     /// Read raw bytes for a blob by its BLAKE3 hash.
 148:     ///
 149:     /// Files < 4 096 bytes are read into a `Vec<u8>`; larger blobs are
 150:     /// memory-mapped for zero-copy access.  Callers dereference the result
 151:     /// to obtain a `&[u8]` slice.
 152:     #[instrument(skip_all)]
 153:     pub fn read_blob(&self, hash: &Blake3Hash) -> Result<CasBytes, CasError> {
 154:         let path = self.blob_path(hash);
 155:         read_cas_bytes(&path)
 156:     }
 157: 
 158:     /// Read a blob by typed id.
 159:     ///
 160:     /// This keeps call-sites in orchestration crates free from raw hash
 161:     /// plumbing while preserving zero-copy read behavior.
 162:     #[instrument(skip_all)]
 163:     pub fn read_blob_typed(&self, id: BlobId) -> Result<CasBytes, CasError> {
 164:         self.read_blob(&id.0)
 165:     }
 166: 
 167:     /// Return `true` when the blob exists in `.arc/blobs/`.
 168:     #[instrument(skip_all)]
 169:     pub fn contains_blob(&self, hash: &Blake3Hash) -> bool {
 170:         self.blob_path(hash).exists()
 171:     }
 172: 
 173:     /// Return the filesystem path where the given blob is stored.
 174:     ///
 175:     /// Useful for callers that need to stream the blob directly from disk
 176:     /// rather than loading it into RAM (e.g. the HTTP push path in
 177:     /// `arc-cli` streams via `PUT /blobs/:hash` without buffering).
 178:     #[instrument(skip_all)]
 179:     pub fn blob_file_path(&self, hash: &Blake3Hash) -> PathBuf {
 180:         self.blob_path(hash)
 181:     }
 182: 
 183:     /// Persist serialized change bytes under a typed change identifier.
 184:     ///
 185:     /// I/O boundary: this method writes to local CAS only. Network transfer,
 186:     /// signature verification, and merge algebra remain outside this crate.
 187:     #[instrument(skip_all)]
 188:     pub fn write_change_bytes(&self, id: ChangeId, bytes: &[u8]) -> Result<ChangeId, CasError> {
 189:         self.write_object(&id.0, bytes)?;
 190:         Ok(id)
 191:     }
 192: 
 193:     /// Read serialized change bytes by typed identifier.
 194:     ///
 195:     /// The returned value preserves zero-copy semantics for large payloads
 196:     /// by using read-only memory maps.
 197:     #[instrument(skip_all)]
 198:     pub fn read_change_bytes(&self, id: ChangeId) -> Result<CasBytes, CasError> {
 199:         self.read_object(&id.0)
 200:     }
 201: }

 203: /// Encode a 32-byte hash as a lowercase hex string (64 chars).
 204: fn hex_encode(hash: &Blake3Hash) -> String {
 205:     hash.iter().fold(String::with_capacity(64), |mut acc, b| {
 206:         use std::fmt::Write;
 207:         let _ = write!(acc, "{b:02x}");
 208:         acc
 209:     })
 210: }
 211: 
 212: /// Read CAS bytes with a small-object fast path and mmap for larger blobs.
 213: ///
 214: /// This mirrors the high-throughput split used in modern object databases:
 215: /// avoid mmap setup overhead for tiny payloads while preserving zero-copy
 216: /// access for larger immutable files.
 217: fn read_cas_bytes(path: &Path) -> Result<CasBytes, CasError> {
 218:     let mut file = fs::File::open(path)?;
 219:     let len = file.metadata()?.len();
 220: 
 221:     if len < SMALL_OBJECT_THRESHOLD {
 222:         let mut buf = Vec::with_capacity(len as usize);
 223:         file.read_to_end(&mut buf)?;
 224:         return Ok(CasBytes::Owned(Bytes::from(buf)));
 225:     }
 226: 
 227:     // SAFETY: CAS files are immutable after successful publish.
 228:     let mapped = unsafe { MmapOptions::new().map_copy_read_only(&file)? };
 229:     Ok(CasBytes::Mapped(mapped))
 230: }

 232: /// Publish bytes atomically into CAS with crash-consistency semantics.
 233: ///
 234: /// The write protocol is:
 235: /// 1. write full payload into a unique temp file in the destination directory,
 236: /// 2. fsync the temp file data,
 237: /// 3. atomically rename into final path,
 238: /// 4. fsync the parent directory when supported by the platform.
 239: ///
 240: /// If another writer already published the same object concurrently, this
 241: /// function treats `AlreadyExists` at rename time as a benign dedup outcome.
 242: fn write_once_atomic(path: &Path, bytes: &[u8]) -> Result<bool, CasError> {
 243:     if path.exists() {
 244:         return Ok(false);
 245:     }
 246: 
 247:     let parent = path.parent().ok_or_else(|| {
 248:         std::io::Error::new(
 249:             std::io::ErrorKind::InvalidInput,
 250:             "CAS object path must have a parent directory",
 251:         )
 252:     })?;
 253:     fs::create_dir_all(parent)?;
 254: 
 255:     let nonce = TEMP_FILE_COUNTER.fetch_add(1, Ordering::Relaxed);
 256:     let file_name = path
 257:         .file_name()
 258:         .unwrap_or_else(|| OsStr::new("cas-object"))
 259:         .to_string_lossy();
 260:     let tmp_path = parent.join(format!(".{file_name}.tmp-{}-{nonce}", std::process::id()));
 261: 
 262:     let mut tmp = fs::OpenOptions::new()
 263:         .create_new(true)
 264:         .write(true)
 265:         .open(&tmp_path)?;
 266:     tmp.write_all(bytes)?;
 267:     tmp.sync_data()?;
 268:     drop(tmp);
 269: 
 270:     match fs::rename(&tmp_path, path) {
 271:         Ok(()) => {
 272:             sync_directory_if_supported(parent)?;
 273:             Ok(true)
 274:         }
 275:         Err(err) if err.kind() == std::io::ErrorKind::AlreadyExists => {
 276:             let _ = fs::remove_file(&tmp_path);
 277:             Ok(false)
 278:         }
 279:         Err(err) => {
 280:             let _ = fs::remove_file(&tmp_path);
 281:             Err(err.into())
 282:         }
 283:     }
 284: }
 285: 
 286: #[cfg(unix)]
 287: fn sync_directory_if_supported(parent: &Path) -> std::io::Result<()> {
 288:     fs::File::open(parent)?.sync_all()
 289: }
 290: 
 291: #[cfg(not(unix))]
 292: fn sync_directory_if_supported(_parent: &Path) -> std::io::Result<()> {
 293:     Ok(())
 294: }
